service cloud.firestore {
  match /databases/{database}/documents {

    function isFamily() {
      return request.auth != null && request.auth.token.email in [
        "smjoshi@gmail.com",
        "bioconcepts.pune@gmail.com",
        "saee1175@gmail.com",
        "gjoshi1208@gmail.com"
      ];
    }
    // Location records: users can write their own, all authenticated can read
    match /locations/{uid} {
      allow read: if isFamily();
      allow write: if request.auth != null && request.auth.uid == uid;

      match /records/{recordId} {
        allow read: if isFamily();
        allow write: if request.auth != null && request.auth.uid == uid;
      }
    }

    // Express sync: any authenticated user can read/write (family broadcast)
    match /express_sync/{docId} {
      allow read, write: if isFamily();
    }
  }
